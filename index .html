01# Two Pointer Optimal Solution - O(N) Time, O(1) Space
02def two_sum_sorted(nums, target):
03 left, right = 0, len(nums) - 1
04 while left < right:
05 total = nums[left] + nums[right]
06 if total == target:
07 return [left + 1, right + 1] # 1-indexed
08 elif total < target:
09 left += 1
10 else:
11 right -= 1
12 return []
